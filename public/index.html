function initResize(e, projId, direction) { e.stopPropagation(); e.preventDefault(); const proj = projects.find(p => p.id === projId); if (!proj) return; const startX = e.clientX; const cellWidth = 40; // Largeur exacte d'une case jour sur la grille (w-10 = 40px) // On mémorise les dates initiales au moment du clic const initialReception = parseLocalDate(proj.date_reception); const targetField = proj.date_reelle ? 'date_reelle' : 'date_estimee'; const initialTargetDate = parseLocalDate(proj[targetField]); function onMouseMove(moveEvent) { const deltaX = moveEvent.clientX - startX; const dayShift = Math.round(deltaX / cellWidth); if (direction === 'left') { let newDate = new Date(initialReception); newDate.setDate(newDate.getDate() + dayShift); // Empêcher de dépasser la date de fin const currentEnd = parseLocalDate(proj.date_reelle || proj.date_estimee); if (newDate <= currentEnd) { proj.date_reception = formatLocalDate(newDate); renderTimeline(); // Mise à jour fluide en direct } } else if (direction === 'right') { let newDate = new Date(initialTargetDate); newDate.setDate(newDate.getDate() + dayShift); // Empêcher de passer avant la date de début const start = parseLocalDate(proj.date_reception); if (newDate >= start) { proj[targetField] = formatLocalDate(newDate); renderTimeline(); // Mise à jour fluide en direct } } } async function onMouseUp() { window.removeEventListener('mousemove', onMouseMove); window.removeEventListener('mouseup', onMouseUp); // Sauvegarde finale en base une fois le glissé-déposé terminé await saveProjectDirect(proj); } window.addEventListener('mousemove', onMouseMove); window.addEventListener('mouseup', onMouseUp); }
